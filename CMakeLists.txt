cmake_minimum_required(VERSION 3.16)
project(journal LANGUAGES CXX)

# Durable record storage over completion I/O: the journal, artifacts, mutations,
# reclamation, inventory, and storage admission. Header-only; no Asio or reactor.
include(FetchContent)
# The coroutine drivers run on task (blocking_scheduler, the wheel). Declared first, so this pin
# wins over the one completion declares (task:18eb516, an ancestor of this pin); no Asio.
FetchContent_Declare(task GIT_REPOSITORY https://github.com/Kronuz/task.git GIT_TAG 477a7d3f15a8951f2047616b834753177d838a8b)
set(TASK_ASIO OFF)
# completion::task: completion_ref, InlineCompletion and HeldCompletion over task.hh.
set(COMPLETION_TASK ON)
set(COMPLETION_ASIO OFF)
FetchContent_Declare(completion GIT_REPOSITORY https://github.com/Kronuz/completion.git GIT_TAG 30615161bebf6adb8297a8a16b338cb1952ce692)
# Fixed-vector reservation accounting shared with managed local storage.
FetchContent_Declare(allocators GIT_REPOSITORY https://github.com/Kronuz/allocators.git GIT_TAG ae775ab995db8e9875e97eae0f01cf634383cd97)
# CRC-32C: one implementation, header-only, no dependencies.
FetchContent_Declare(crc32c GIT_REPOSITORY https://github.com/Kronuz/crc32c.git GIT_TAG 52889db30c9902c3ca42c932a4e6f4ca4f8da87b)
FetchContent_MakeAvailable(completion allocators crc32c)

add_library(journal INTERFACE)
target_include_directories(journal INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(journal INTERFACE cxx_std_20)
target_link_libraries(journal INTERFACE completion::task allocators::resources crc32c::crc32c)
add_library(journal::journal ALIAS journal)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	enable_testing()

		endif()
		if(NOT t STREQUAL "placement" AND NOT t STREQUAL "turn_seal" AND NOT t STREQUAL "host" AND NOT t STREQUAL "host_delete")

	endforeach()
	target_sources(journal_host_test PRIVATE test/host_idle_test.cc)  # host_test's idle-group cases

		add_test(NAME journal_host_${c} COMMAND journal_host_test ${c})
		set_tests_properties(journal_host_${c} PROPERTIES TIMEOUT 600)
	endforeach()
	foreach(c reopen_empty stays_deleted crash crash_finish refused loop_thread)

	foreach(c crash sequence budget full_size staged_room bound forget_in_flight forget_from_sealer forget_threads trigger supersede unseal_owner lock_order wake_threads wake_refused wake_after_run unseal_threads repeat repeat_turns repeat_throw repeat_latent repeat_unseal repeat_threads)
