cmake_minimum_required(VERSION 3.16)
project(journal LANGUAGES CXX)

# Durable record storage over completion I/O: the journal, artifacts, mutations,
# reclamation, inventory, and storage admission. Header-only; no Asio or reactor.
include(FetchContent)
FetchContent_Declare(completion GIT_REPOSITORY https://github.com/Kronuz/completion.git GIT_TAG 373f02a219a91e32af035abf6fb909a8211aef8b)
FetchContent_MakeAvailable(completion)

add_library(journal INTERFACE)
target_include_directories(journal INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_compile_features(journal INTERFACE cxx_std_20)
target_link_libraries(journal INTERFACE completion::completion)
add_library(journal::journal ALIAS journal)

if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
	enable_testing()
	file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/.scratch")  # the tests' state
	foreach(t journal inventory admission turn_log)
		add_executable(journal_${t}_test test/${t}_test.cc)
		target_link_libraries(journal_${t}_test PRIVATE journal::journal)
		add_test(NAME journal_${t} COMMAND journal_${t}_test)
		set_tests_properties(journal_${t} PROPERTIES TIMEOUT 30)
	endforeach()
	if(APPLE OR CMAKE_SYSTEM_NAME STREQUAL "FreeBSD" OR CMAKE_SYSTEM_NAME STREQUAL "Linux")
		add_executable(journal_completion_test test/completion_test.cc)
		target_link_libraries(journal_completion_test PRIVATE journal::journal)
		add_test(NAME journal_completion COMMAND journal_completion_test)
		set_tests_properties(journal_completion PROPERTIES TIMEOUT 30)
	endif()
	if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
		add_executable(journal_linux_completion_test test/linux_completion_test.cc)
		target_link_libraries(journal_linux_completion_test PRIVATE journal::journal)
		add_test(NAME journal_linux_completion COMMAND journal_linux_completion_test)
		set_tests_properties(journal_linux_completion PROPERTIES TIMEOUT 30)
		add_executable(journal_completion_bench bench/completion_bench.cc)
		target_link_libraries(journal_completion_bench PRIVATE journal::journal)
	endif()
	foreach(b bench checkpoint_bench reclaim_bench)
		add_executable(journal_${b} bench/${b}.cc)
		target_link_libraries(journal_${b} PRIVATE journal::journal)
	endforeach()
	add_executable(journal_demo examples/demo.cc)
	target_link_libraries(journal_demo PRIVATE journal::journal)
endif()
